delivery_optimizer.py — Dashly Unit Economics DRF Engine 
def calculate_order_margin(basket: float, markup_pct: float, cogs_pct: float):
    gross = basket * (1 + markup_pct / 100.0)
    dashly_cut = gross * 0.33       # 33% Commission
    packaging = 0.45                # £0.45 Fixed packaging
    labor = 0.96                    # 4 mins @ 1.13x employer on-cost
    cogs = basket * (cogs_pct / 100.0)
    net = gross - (dashly_cut + packaging + labor + cogs)
    return {"gross": gross, "net_profit": net, "margin_pct": (net / gross) * 100}
